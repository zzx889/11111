function gradient(text, startColor, endColor)
    local result = ""; local chars = {}
    for uchar in text:gmatch("[%z\1-\127\194-\244][\128-\191]*") do table.insert(chars, uchar) end
    local length = #chars
    for i = 1, length do
        local t = (i - 1) / math.max(length - 1, 1)
        local r = startColor.R + (endColor.R - startColor.R) * t
        local g = startColor.G + (endColor.G - startColor.G) * t
        local b = startColor.B + (endColor.B - startColor.B) * t
        result = result .. string.format('<font color="rgb(%d,%d,%d)">%s</font>', math.floor(r * 255), math.floor(g * 255), math.floor(b * 255), chars[i])
    end
    return result
end

    -- 内建飞行
    local BuiltInFlyEnabled = false
    local flyBodyGyro = nil
    local flyBodyVel = nil
    local function startBuiltInFly()
        if BuiltInFlyEnabled then return end
        BuiltInFlyEnabled = true
        local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        local root = char:WaitForChild("HumanoidRootPart")
        local hum = char:WaitForChild("Humanoid")
        root.Anchored = false
        flyBodyGyro = Instance.new("BodyGyro", root)
        flyBodyGyro.MaxTorque = Vector3.new(1/0,1/0,1/0)
        flyBodyVel = Instance.new("BodyVelocity", root)
        flyBodyVel.MaxForce = Vector3.new(1/0,1/0,1/0)
        flyBodyVel.Velocity = Vector3.zero
        local keys = {}
        UserInputService.InputBegan:Connect(function(input, gpe)
            if not BuiltInFlyEnabled or gpe then return end
            keys[input.KeyCode] = true
        end)
        UserInputService.InputEnded:Connect(function(input)
            keys[input.KeyCode] = nil
        end)
        RunService.RenderStepped:Connect(function()
            if not BuiltInFlyEnabled then
                if flyBodyGyro then flyBodyGyro:Destroy() end
                if flyBodyVel then flyBodyVel:Destroy() end
                return
            end
            local dir = Vector3.zero
            if keys[Enum.KeyCode.W] then dir += Camera.CFrame.LookVector end
            if keys[Enum.KeyCode.S] then dir -= Camera.CFrame.LookVector end
            if keys[Enum.KeyCode.A] then dir -= Camera.CFrame.RightVector end
            if keys[Enum.KeyCode.D] then dir += Camera.CFrame.RightVector end
            if keys[Enum.KeyCode.E] then dir += Vector3.new(0,1,0) end
            if keys[Enum.KeyCode.Q] then dir -= Vector3.new(0,1,0) end
            if dir.Magnitude > 0 then dir = dir.Unit * 50 end
            flyBodyVel.Velocity = dir
            flyBodyGyro.CFrame = Camera.CFrame
        end)
    end
    local function stopBuiltInFly()
        BuiltInFlyEnabled = false
        if flyBodyGyro then flyBodyGyro:Destroy() end
        if flyBodyVel then flyBodyVel:Destroy() end
    end

        end
    end

    local function exportConfig()
        local config = {
            Speed = TargetWalkSpeed,
            Jump = CustomJumpValue,
            FOV = FOV,
            ESP = ESP_SETTINGS,
            PLAYER_ESP = PLAYER_ESP,
            Spin = SpinSpeed,
            LootKeywords = LootKeywords,
            PickupRange = PickupRange,
            BulletSpeed = BulletSpeed,
            MorphSize = MorphSize,
            ExplosionPower = ExplosionPower
        }
        local json = HttpService:JSONEncode(config)
        if setclipboard then setclipboard(json); Notify("配置已复制到剪贴板", "可发送到其他设备导入", 3)
        else Notify("导出失败", "剪贴板不可用", 2) end
    end
    local function importConfig()
        if not setclipboard then Notify("导入失败", "剪贴板不可用", 2) return end
        local input = getclipboard and getclipboard() or ""
        if input == "" then Notify("剪贴板为空", "", 2) return end
        pcall(function()
            local config = HttpService:JSONDecode(input)
            if config.Speed then TargetWalkSpeed = config.Speed end
            if config.Jump then CustomJumpValue = config.Jump end
            if config.FOV then FOV = config.FOV end
            if config.ESP then ESP_SETTINGS = config.ESP end
            if config.PLAYER_ESP then PLAYER_ESP = config.PLAYER_ESP end
            if config.Spin then SpinSpeed = config.Spin end
            if config.LootKeywords then LootKeywords = config.LootKeywords; refreshLootESP() end
            if config.PickupRange then PickupRange = config.PickupRange end
            if config.BulletSpeed then BulletSpeed = config.BulletSpeed end
            if config.MorphSize then MorphSize = Vector3.new(config.MorphSize.X, config.MorphSize.Y, config.MorphSize.Z); applyMorph() end
            if config.ExplosionPower then ExplosionPower = config.ExplosionPower end
            Notify("配置导入成功", "已应用所有设置", 3)
        end)
    end

    -- ==================== 翻译功能 ====================
    -- 作者: TX (改编自 TX自动翻译器)
    local Translator = {
        Enabled = false,
        TargetLang = "zh-CN",
        MaxTextLength = 500,
        BaseScanInterval = 1.5,
        SpeedMultiplier = 2,
        HeartbeatConnection = nil,
        TranslationCache = {},
        TranslatedInstances = {},
        EmoteKeywords = {"neon", "shine", "ghost", "gold", "spin", "bighead", "smallhead", "giant", "squash"},
    }

    function Translator.detectLanguage(text)
        if not text or #text == 0 then return "en" end
        local counts = { cjk = 0, zht = 0, ko = 0, ja = 0, en = 0 }
        for _, code in utf8.codes(text) do
            local char = utf8.char(code)
            if char:match("[\228-\233][\128-\191][\128-\191]") then counts.cjk = counts.cjk + 1
            elseif char:match("[\227][\128-\191][\128-\191]") then counts.zht = counts.zht + 1
            elseif char:match("[\234-\237][\128-\191]") then counts.ko = counts.ko + 1
            elseif char:match("[\227-\227][\128-\191][\128-\191]*") or char:match("[\227-\233][\128-\191]") then counts.ja = counts.ja + 1
            elseif char:match("[%a]") then counts.en = counts.en + 1 end
        end
        if counts.cjk > 3 then return counts.zht > 3 and "zh-TW" or "zh-CN" end
        if counts.ko > 2 then return "ko" end
        if counts.ja > 2 then return "ja" end
        if counts.en > 5 then return "en" end
        return "en"
    end

    function Translator.shouldSkipText(text)
        if not text or text == "" or Translator.TranslationCache[text] then return true end
        if text:match("^%s*$") then return true end
        if text:match("^[%d%.%%,%s:/]+$") then return true end
        if #text > Translator.MaxTextLength then return true end
        if Players:FindFirstChild(text) then return true end
        local lower = string.lower(text)
        for _, kw in ipairs(Translator.EmoteKeywords) do
            if string.find(lower, kw) then return true end
        end
        return false
    end

    function Translator.translateGoogle(text, fromLang, toLang)
        local url = string.format(
            "https://translate.googleapis.com/translate_a/single?client=gtx&sl=%s&tl=%s&dt=t&q=%s",
            fromLang, toLang, HttpService:UrlEncode(text)
        )
        local ok, body = pcall(game.HttpGet, game, url, false, {["User-Agent"] = "Mozilla/5.0"})
        if not ok or not body then return nil end
        local s, data = pcall(HttpService.JSONDecode, HttpService, body)
        if not s or not data or not data[1] then return nil end
        local result = ""
        for _, part in ipairs(data[1]) do
            if part[1] then result = result .. part[1] end
        end
        return result ~= "" and result or nil
    end

    function Translator.translateMyMemory(text, fromLang, toLang)
        local url = string.format(
            "https://api.mymemory.translated.net/get?q=%s&langpair=%s|%s",
            HttpService:UrlEncode(text), fromLang, toLang
        )
        local ok, body = pcall(game.HttpGet, game, url, false, {["User-Agent"] = "Roblox"})
        if not ok or not body then return nil end
        local s, data = pcall(HttpService.JSONDecode, HttpService, body)
        if not s or not data or not data.responseData then return nil end
        return data.responseData.translatedText
    end

    function Translator.translateText(text)
        if Translator.shouldSkipText(text) then return Translator.TranslationCache[text] or text end
        local detected = Translator.detectLanguage(text)
        if detected == "zh-CN" then return text end
        local translated = Translator.translateGoogle(text, detected, Translator.TargetLang)
        if not translated then translated = Translator.translateMyMemory(text, detected, Translator.TargetLang) end
        if translated then
            Translator.TranslationCache[text] = translated
            return translated
        end
        return text
    end

    local function isTextInstance(obj)
        return obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")
    end

    local function shouldSkipInstance(obj)
        if not obj then return true end
        if obj.Parent and obj.Parent:IsA("TextChannel") then return true end
        return false
    end

    function Translator.translateInstance(instance)
        if not isTextInstance(instance) or Translator.TranslatedInstances[instance] then return false end
        if shouldSkipInstance(instance) then return false end
        local original = instance.Text
        if not original or original == "" then return false end
        Translator.TranslatedInstances[instance] = true
        local newText = Translator.translateText(original)
        if newText ~= original then
            instance.Text = newText
            return true
        end
        return false
    end

    local lastScanTime = 0
    function Translator.scanAndTranslate(root)
        local now = tick()
        if now - lastScanTime < Translator.BaseScanInterval / Translator.SpeedMultiplier then
            return 0
        end
        lastScanTime = now
        local count = 0
        for _, descendant in ipairs(root:GetDescendants()) do
            if isTextInstance(descendant) and not Translator.TranslatedInstances[descendant] and not shouldSkipInstance(descendant) then
                if Translator.translateInstance(descendant) then
                    count = count + 1
                end
            end
        end
        return count
    end

    function Translator.start()
        if Translator.Enabled then return end
        Translator.Enabled = true
        local gui = LocalPlayer:WaitForChild("PlayerGui")
        Translator.scanAndTranslate(gui)
        if Translator.HeartbeatConnection then
            Translator.HeartbeatConnection:Disconnect()
        end
        Translator.HeartbeatConnection = RunService.Heartbeat:Connect(function()
            if Translator.Enabled then
                Translator.scanAndTranslate(gui)
            end
        end)
        gui.DescendantAdded:Connect(function(obj)
            if Translator.Enabled and isTextInstance(obj) then
                task.delay(0.1, function()
                    if obj.Parent and not shouldSkipInstance(obj) then
                        Translator.translateInstance(obj)
                    end
                end)
            end
        end)
        local core = game:GetService("CoreGui")
        Translator.scanAndTranslate(core)
        core.DescendantAdded:Connect(function(obj)
            if Translator.Enabled and isTextInstance(obj) then
                task.delay(0.1, function()
                    if obj.Parent and not shouldSkipInstance(obj) then
                        Translator.translateInstance(obj)
                    end
                end)
            end
        end)
    end

    function Translator.stop()
        Translator.Enabled = false
        if Translator.HeartbeatConnection then
            Translator.HeartbeatConnection:Disconnect()
            Translator.HeartbeatConnection = nil
        end
    end
